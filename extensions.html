進階✓ arduino-cli 編譯通過（310 KB）

兩顆加速度計掛同一條 I2C（0x53／0x68），50 Hz 同步讀、各跑一份相同的計步演算法、按左鍵做六面校正；電腦端腳本算偏移、雜訊、相關係數、步數一致性並畫圖。

  1. 兩顆模組用雙面膠貼在同一塊小板上，軸向對齊；VCC→3V3、GND→GND、SDA→21、SCL→22；MPU6050 的 AD0 不接（0x68）。
  2. make ext NAME=dual_sensor_compare → make flash-ext NAME=dual_sensor_compare PORT=…。
  3. python3 tools/compare_sensors.py /dev/cu.usbserial-XXXX --secs 60：先靜置 10 秒、走 30 秒、再靜置。
  4. 要校正：在韌體端按左鍵，依序六面朝下各按一次，再收一次資料比對校正前後。

// 延伸專案 A：雙感測器比對（ADXL345 @0x53 ＋ MPU6050 @0x68 掛同一條 I2C）
// 50 Hz 同步讀兩顆，序列埠輸出 CSV；兩顆各跑一份同樣的計步演算法；按下鍵（GPIO0）做六面校正。
// 接線：兩顆模組的 VCC→3V3、GND→GND、SDA→GPIO21、SCL→GPIO22；MPU6050 的 AD0 接 GND（或不接）→ 位址 0x68。
// 兩顆模組請用雙面膠貼在同一塊小板上，X/Y/Z 方向一致，才比得出差異。
// 電腦端：python3 tools/compare_sensors.py /dev/cu.usbserial-XXXX
#include <Wire.h>
#include "pedometer.h"      // 與主韌體同一份演算法（Makefile 會複製進來）

static const uint8_t ADXL = 0x53, MPU = 0x68;
static const int PIN_BTN_L = 0, PIN_BTN_R = 35, PIN_SDA = 21, PIN_SCL = 22;

struct Vec { float x, y, z; };
struct Cal { Vec off{0, 0, 0}; Vec gain{1, 1, 1}; };     // 校正：a_cal = (a_raw − off) × gain
static Cal cal1, cal2;
static Pedometer pedo1, pedo2;

static void wr(uint8_t a, uint8_t r, uint8_t v) { Wire.beginTransmission(a); Wire.write(r); Wire.write(v); Wire.endTransmission(); }
static bool rd(uint8_t a, uint8_t r, uint8_t* b, uint8_t n) {
  Wire.beginTransmission(a); Wire.write(r);
  if (Wire.endTransmission(false) != 0) return false;
  return Wire.requestFrom(a, n) == n && [&] { for (uint8_t i = 0; i < n; i++) b[i] = Wire.read(); return true; }();
}
static bool adxl_begin() {
  uint8_t id; if (!rd(ADXL, 0x00, &id, 1) || id != 0xE5) return false;
  wr(ADXL, 0x2C, 0x0A); wr(ADXL, 0x31, 0x08); wr(ADXL, 0x2D, 0x08);  // 100 Hz、全解析度 ±2g、量測
  return true;
}
static bool mpu_begin() {
  uint8_t id; if (!rd(MPU, 0x75, &id, 1) || (id != 0x68 && id != 0x70 && id != 0x71)) return false;  // WHO_AM_I（6050=0x68，6500/9250 相容）
  wr(MPU, 0x6B, 0x00);   // PWR_MGMT_1：喚醒
  wr(MPU, 0x1C, 0x00);   // ACCEL_CONFIG：±2g → 16384 LSB/g
  wr(MPU, 0x1A, 0x03);   // DLPF 約 44 Hz
  return true;
}
static bool adxl_read(Vec& v) { uint8_t b[6]; if (!rd(ADXL, 0x32, b, 6)) return false;
  v.x = (int16_t)(b[0] | b[1] << 8) / 256.0f; v.y = (int16_t)(b[2] | b[3] << 8) / 256.0f; v.z = (int16_t)(b[4] | b[5] << 8) / 256.0f; return true; }
static bool mpu_read(Vec& v) { uint8_t b[6]; if (!rd(MPU, 0x3B, b, 6)) return false;
  v.x = (int16_t)(b[0] << 8 | b[1]) / 16384.0f; v.y = (int16_t)(b[2] << 8 | b[3]) / 16384.0f; v.z = (int16_t)(b[4] << 8 | b[5]) / 16384.0f; return true; }
static Vec apply(const Vec& a, const Cal& c) { return { (a.x - c.off.x) * c.gain.x, (a.y - c.off.y) * c.gain.y, (a.z - c.off.z) * c.gain.z }; }

// 六面校正：依序把板子的 +X、−X、+Y、−Y、+Z、−Z 朝下靜置，每面按一次下鍵（GPIO0），各取 100 筆平均
static void six_face_calibration() {
  const char* faces[] = { "+X 朝下", "-X 朝下", "+Y 朝下", "-Y 朝下", "+Z 朝下", "-Z 朝下" };
  Vec s1[6], s2[6];
  for (int f = 0; f < 6; f++) {
    Serial.printf("# 校正 %d/6：請把板子 %s 靜置，然後按下鍵（GPIO0）\n", f + 1, faces[f]);
    while (digitalRead(PIN_BTN_L) == HIGH) delay(10);
    delay(300);
    Vec a{0, 0, 0}, b{0, 0, 0}; int n = 0;
    for (int i = 0; i < 100; i++) { Vec v1, v2; if (adxl_read(v1) && mpu_read(v2)) { a.x += v1.x; a.y += v1.y; a.z += v1.z; b.x += v2.x; b.y += v2.y; b.z += v2.z; n++; } delay(10); }
    s1[f] = { a.x / n, a.y / n, a.z / n }; s2[f] = { b.x / n, b.y / n, b.z / n };
    while (digitalRead(PIN_BTN_L) == LOW) delay(10);
  }
  auto fit = [](Vec* s, Cal& c) {   // 每軸：最大與最小的平均是偏移，差值的一半是 1 g 的刻度
    float xs[2] = { s[0].x, s[1].x }, ys[2] = { s[2].y, s[3].y }, zs[2] = { s[4].z, s[5].z };
    auto one = [](float p, float m, float& off, float& g) { off = (p + m) / 2; g = 2.0f / fabsf(p - m); };
    one(xs[0], xs[1], c.off.x, c.gain.x); one(ys[0], ys[1], c.off.y, c.gain.y); one(zs[0], zs[1], c.off.z, c.gain.z);
  };
  fit(s1, cal1); fit(s2, cal2);
  Serial.printf("# CAL ADXL345 off=%.4f,%.4f,%.4f gain=%.4f,%.4f,%.4f\n", cal1.off.x, cal1.off.y, cal1.off.z, cal1.gain.x, cal1.gain.y, cal1.gain.z);
  Serial.printf("# CAL MPU6050 off=%.4f,%.4f,%.4f gain=%.4f,%.4f,%.4f\n", cal2.off.x, cal2.off.y, cal2.off.z, cal2.gain.x, cal2.gain.y, cal2.gain.z);
  Serial.println("# 校正完成，繼續輸出 CSV（已套用校正）");
}

void setup() {
  Serial.begin(115200); delay(200);
  pinMode(PIN_BTN_L, INPUT_PULLUP); pinMode(PIN_BTN_R, INPUT);
  Wire.begin(PIN_SDA, PIN_SCL, 400000);
  bool a = adxl_begin(), m = mpu_begin();
  Serial.printf("# ADXL345 %s / MPU6050 %s\n", a ? "OK" : "找不到", m ? "OK" : "找不到");
  Serial.println("# 下鍵（GPIO0）：六面校正　上鍵（GPIO35）：兩顆步數歸零");
  Serial.println("t_ms,a1x,a1y,a1z,a2x,a2y,a2z,steps1,steps2");
  pedo1.begin(0); pedo2.begin(0);
}

void loop() {
  static uint32_t next = 0; uint32_t now = millis();
  if (digitalRead(PIN_BTN_L) == LOW) { while (digitalRead(PIN_BTN_L) == LOW) delay(10); six_face_calibration(); }
  if (digitalRead(PIN_BTN_R) == LOW) { pedo1.reset(); pedo2.reset(); Serial.println("# 步數歸零"); delay(300); }
  if (now < next) return; next = now + 20;                         // 50 Hz
  Vec r1, r2; if (!adxl_read(r1) || !mpu_read(r2)) return;
  Vec v1 = apply(r1, cal1), v2 = apply(r2, cal2);
  pedo1.feed(v1.x, v1.y, v1.z, now); pedo2.feed(v2.x, v2.y, v2.z, now);
  Serial.printf("%u,%.4f,%.4f,%.4f,%.4f,%.4f,%.4f,%u,%u\n", now, v1.x, v1.y, v1.z, v2.x, v2.y, v2.z, pedo1.steps(), pedo2.steps());
}

// 延伸專案 B：久坐提醒（原作說明書寫了但沒做的功能）
// 純 C++，不碰硬體。用法（在 app.cpp）：
//   static Sedentary sed;                         // 全域
//   在 app_loop() 每次取樣後：sed.feed(pedo.steps(), now);
//   if (sed.due(now)) { buzz.play(MELODY_SEDENTARY, 4); hal_log("[久坐] 已 %u 分鐘沒走動", sed.idle_minutes(now)); }
//   番茄鐘進行中不提醒：sed.suspend(pomo.state() == POMO_RUNNING);
// 測試：clang++ -std=c++17 firmware/extensions/sedentary/test_sedentary.cpp -o build/test_sedentary && build/test_sedentary
#pragma once
#include <stdint.h>

class Sedentary {
public:
  // limit_min：連續幾分鐘沒有步數就提醒；repeat_min：提醒後若仍不動，隔幾分鐘再提醒一次
  explicit Sedentary(uint16_t limit_min = 60, uint16_t repeat_min = 15) : limit_(limit_min), repeat_(repeat_min) {}
  void feed(uint32_t steps, uint32_t now_ms) {
    if (!primed_) { last_steps_ = steps; last_move_ = now_ms; primed_ = true; return; }
    if (steps != last_steps_) { last_steps_ = steps; last_move_ = now_ms; reminded_ = 0; }
  }
  void suspend(bool s) { suspended_ = s; }
  uint32_t idle_minutes(uint32_t now_ms) const { return primed_ ? (now_ms - last_move_) / 60000u : 0; }
  // 到了該提醒的時刻回傳 true（每個時刻只回一次）
  bool due(uint32_t now_ms) {
    if (!primed_ || suspended_) return false;
    uint32_t idle = idle_minutes(now_ms);
    uint32_t next = reminded_ == 0 ? limit_ : limit_ + reminded_ * repeat_;
    if (idle >= next) { reminded_++; return true; }
    return false;
  }
  void set_limits(uint16_t limit_min, uint16_t repeat_min) { limit_ = limit_min; repeat_ = repeat_min; }
private:
  uint16_t limit_, repeat_; uint32_t last_steps_ = 0, last_move_ = 0; uint8_t reminded_ = 0; bool primed_ = false, suspended_ = false;
};

// 做法：在原本「峰值→間隔→連續確認」之外，再要求最近 N 個間隔的變異係數（標準差／平均）低於 cv_max 才開始計。

// 測試：clang++ -std=c++17 -I firmware/app firmware/extensions/pedometer_stable/test_pedometer_stable.cpp firmware/app/pedometer.cpp -o build/test_pedometer_stable && build/test_pedometer_stable
#pragma once
#include <stdint.h>
#include <math.h>

struct StableParams {
  float    k_thresh      = 0.12f;
  uint16_t min_interval  = 250;
  uint16_t max_interval  = 2000;
  uint8_t  confirm_steps = 4;      // 要 4 個峰值才有 3 個間隔可以算變異
  float    cv_max        = 0.25f;  // 間隔變異係數上限：0.25 ＝ 間隔忽快忽慢超過 25% 就不算走路
};

class PedometerStable {
public:
  void begin(uint32_t saved) { steps_ = saved; }
  void feed(float ax, float ay, float az, uint32_t now_ms) {
    float m = sqrtf(ax * ax + ay * ay + az * az);
    if (!primed_) { lp_ = dc_ = m; primed_ = true; return; }
    lp_ += 0.3f * (m - lp_); dc_ += 0.02f * (lp_ - dc_);
    prev2_ = prev_; prev_ = s_; s_ = lp_ - dc_;
    bool peak = prev_ > prev2_ && prev_ >= s_ && prev_ > p.k_thresh;
    if (!peak) { if (pending_ && now_ms - last_ > p.max_interval) pending_ = 0; return; }
    uint32_t gap = now_ms - last_;
    if (last_ && gap < p.min_interval) return;
    if (last_ == 0 || gap > p.max_interval) { pending_ = 0; last_ = now_ms; pending_ = 1; return; }
    last_ = now_ms;
    gaps_[gap_i_++ % 3] = gap;                       // 保留最近 3 個間隔
    if (pending_ < p.confirm_steps) {
      pending_++;
      if (pending_ >= p.confirm_steps) {
        if (cv_ok()) { steps_ += pending_; confirmed_ = true; } else pending_ = 1;   // 節奏亂：重新累積
      }
    } else {
      if (cv_ok()) steps_++; else { pending_ = 1; confirmed_ = false; }
    }
  }
  uint32_t steps() const { return steps_; }
  void reset() { steps_ = 0; pending_ = 0; }
  void set_steps(uint32_t s) { steps_ = s; }
  float signal() const { return s_; }
  StableParams p;
private:
  bool cv_ok() const {
    if (gap_i_ < 3) return false;
    float mean = (gaps_[0] + gaps_[1] + gaps_[2]) / 3.0f, var = 0;
    for (int i = 0; i < 3; i++) var += (gaps_[i] - mean) * (gaps_[i] - mean);
    return sqrtf(var / 3) / mean <= p.cv_max;
  }
  uint32_t steps_ = 0, last_ = 0, gaps_[3] = {0, 0, 0}; uint8_t gap_i_ = 0, pending_ = 0;
  float lp_ = 1, dc_ = 1, s_ = 0, prev_ = 0, prev2_ = 0; bool primed_ = false, confirmed_ = false;
};

static const BitmapFont* font_of(int f) { return f == FONT_S ? &FONT12 : f == FONT_L ? &FONT24 : &FONT16; }
static const Glyph* find_glyph(const BitmapFont* F, uint32_t cp) {
  int lo = 0, hi = F->n - 1;                                  // glyphs 依 codepoint 排序，二分搜尋
  while (lo <= hi) { int m = (lo + hi) / 2; if (F->glyphs[m].cp == cp) return &F->glyphs[m]; if (F->glyphs[m].cp < cp) lo = m + 1; else hi = m - 1; }
  return cp == '?' ? nullptr : find_glyph(F, '?');            // 沒有的字畫成問號
}
static uint32_t utf8_next(const char** p) {
  const uint8_t* s = (const uint8_t*)*p; uint32_t c = s[0]; int n = 0;
  if (c >= 0xF0) { c &= 7; n = 3; } else if (c >= 0xE0) { c &= 15; n = 2; } else if (c >= 0xC0) { c &= 31; n = 1; }
  for (int i = 1; i <= n && s[i]; i++) c = (c << 6) | (s[i] & 63);
  *p += n + 1; return c;
}
static int text_width(const BitmapFont* F, const char* s) {
  int w = 0; while (*s) { const Glyph* g = find_glyph(F, utf8_next(&s)); if (g) w += g->adv; } return w;
}
static void draw_glyph(const BitmapFont* F, const Glyph* g, int x, int y, uint16_t c) {
  int rowbytes = (g->w + 7) / 8; const uint8_t* b = F->bits + g->off;
  for (int r = 0; r < F->height; r++, b += rowbytes)
    for (int px = 0; px < g->w; px++)
      if (b[px >> 3] & (0x80 >> (px & 7))) canvas.drawPixel(x + px, y + r, c);
}

    if (rot_frame) { rot_frame->deleteSprite(); delete rot_frame; rot_frame = nullptr; }

      // 輕量雜湊：每個 32 位元字「左旋 5 位再 XOR」（ROM 的 crc32 要約 12 ms，這個約 1 ms；UI 畫面碰撞機率可忽略）

      if (c == last_crc) { pf_skip++; return; }
      last_crc = c;
    }
    pf_push++; canvas.pushSprite(0, 0); return;
  }
  if (hw_rot != 1) { tft.setRotation(1); hw_rot = 1; }
  if (!rot_frame) {
    rot_frame = new LGFX_Sprite(&tft); rot_frame->setColorDepth(8);

}

  adxl_write(0x31, 0x08);   // DATA_FORMAT：全解析度、±2g → 3.9 mg/LSB

  adxl_write(0x1D, 40);     // THRESH_TAP：62.5 mg/LSB → 2.5 g

  adxl_write(0x22, 64);     // LATENT：1.25 ms/LSB → 80 ms
  adxl_write(0x23, 240);    // WINDOW：1.25 ms/LSB → 300 ms
  adxl_write(0x2A, 0x07);   // TAP_AXES：X、Y、Z 都偵測
  adxl_write(0x2E, 0x20);   // INT_ENABLE：DOUBLE_TAP（腳沒接，只是讓 INT_SOURCE 會記錄）
  adxl_write(0x38, 0x80 | 31);   // FIFO_CTL：串流模式，存最近 32 筆（100 Hz → 320 ms）
  adxl_write(0x2D, 0x08);   // POWER_CTL：量測模式
  return true;
}

int hal_accel_read_batch(float* xyz, int max, uint8_t* events) {
  *events = 0;
  uint8_t src = adxl_read8(0x30);                     // INT_SOURCE（讀了就清除）
  if (src & 0x20) { *events |= 1; pf_taps++; }
  int n = adxl_read8(0x39) & 0x3F;                    // FIFO_STATUS：目前存了幾筆
  pf_fifo_reads++; pf_fifo_total++; if (n >= 32) pf_fifo_full++; if (n > pf_fifo_max) pf_fifo_max = n;
  int out = 0, got = 0;
  for (int i = 0; i < n; i++) {
    Wire.beginTransmission(ADXL_ADDR); Wire.write(0x32);
    if (Wire.endTransmission(false) != 0 || Wire.requestFrom(ADXL_ADDR, (uint8_t)6) != 6) { pf_fifo_err++; break; }
    got++;
    uint8_t b[6]; for (int k = 0; k < 6; k++) b[k] = Wire.read();
    int16_t x = (int16_t)(b[0] | (b[1] << 8)), y = (int16_t)(b[2] | (b[3] << 8)), z = (int16_t)(b[4] | (b[5] << 8));
    if (!half_have) { half_x = x; half_y = y; half_z = z; half_have = true; continue; }
    half_have = false;
    int16_t mx = (half_x + x) / 2, my = (half_y + y) / 2, mz = (half_z + z) / 2;
    rec_push(mx, my, mz);
    if (out < max) { xyz[out * 3] = mx / 256.0f; xyz[out * 3 + 1] = my / 256.0f; xyz[out * 3 + 2] = mz / 256.0f; out++; }
  }
  pf_fifo_last_n = n; pf_fifo_last_got = got; pf_fifo_last_out = out;
  return out;
}

    ch_batt = bs->createCharacteristic(NimBLEUUID((uint16_t)0x2A19), NIMBLE_PROPERTY::READ | NIMBLE_PROPERTY::NOTIFY);
    bs->start();
    NimBLEService* ps = ble_srv->createService(PED_SVC);
    ch_steps = ps->createCharacteristic(PED_STEPS, NIMBLE_PROPERTY::READ | NIMBLE_PROPERTY::NOTIFY);
    ch_hours = ps->createCharacteristic(PED_HOURS, NIMBLE_PROPERTY::READ);
    NimBLECharacteristic* ch_time = ps->createCharacteristic(PED_TIME, NIMBLE_PROPERTY::WRITE); ch_time->setCallbacks(new TimeCb());
    ch_info = ps->createCharacteristic(PED_INFO, NIMBLE_PROPERTY::READ);
    ps->start();
    NimBLEAdvertising* ad = NimBLEDevice::getAdvertising();
    ad->setName(name); ad->addServiceUUID(PED_SVC); ad->enableScanResponse(true);
    ad->setMinInterval(1600); ad->setMaxInterval(1600);   // 0.625 ms × 1600 ＝ 1 秒廣播一次（省電）

    hal_log("[藍牙] 初始化完成：記憶體 %u → %u（最大連續 %u）", h0, ESP.getFreeHeap(), ESP.getMaxAllocHeap());
  }
  NimBLEDevice::getAdvertising()->start(); ble_on = true; ble_last_steps = 0xFFFFFFFF; ble_last_batt = 255; ble_since = millis();

}

static String rebuild_body() {        // WebServer 已把表單拆成 args，拼回 k=v&k=v 給 app 解析
  String b;
  for (int i = 0; i < server.args(); i++) {
    if (server.argName(i) == "plain") continue;
    if (b.length()) b += '&';
    b += server.argName(i); b += '=';
    String v = server.arg(i);
    for (size_t j = 0; j < v.length(); j++) {     // 只轉 & = % + 四個字元，其餘原樣（app 會解 %XX）
      char c = v[j];
      if (c == '&' || c == '=' || c == '%' || c == '+') { char h[4]; snprintf(h, 4, "%%%02X", (uint8_t)c); b += h; } else b += c;
    }
  }
  if (!b.length() && server.hasArg("plain")) b = server.arg("plain");
  return b;
}

}
// ---------- OTA（Wi-Fi 更新）：GET /update 給表單；POST /update 收 multipart 的 .bin；OPTIONS 給 IDE 頁的 CORS 預檢
static const char UPDATE_FORM[] PROGMEM =
  "<!doctype html><html lang=zh-Hant><meta charset=utf-8><meta name=viewport content='width=device-width,initial-scale=1'><title>韌體更新</title>"
  "<body style='font-family:sans-serif;padding:16px;max-width:520px;line-height:1.7'><h2>韌體更新（OTA）</h2>"
  "<p>選 <code>pedometer_lab.ino.bin</code>（IDE 頁「存檔並編譯」產生，或網站 firmware/app.bin），按「更新」。約 30～60 秒，完成後自動重開機；設定與步數都在 NVS，不會消失。</p>"
  "<form method=POST action=/update enctype=multipart/form-data><input type=file name=fw accept=.bin required><br><br><button style='padding:8px 16px'>更新</button></form>"
  "<p style='color:#666'>帳號 admin，密碼同設定頁。</p></body></html>";
static bool ota_routes_done = false, ota_auth_fail = false;
static void cors() { server.sendHeader("Access-Control-Allow-Origin", "*"); server.sendHeader("Access-Control-Allow-Headers", "Authorization, Content-Type"); server.sendHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); }
static void ota_routes() {
  if (ota_routes_done) return; ota_routes_done = true;
  server.on("/update", HTTP_GET, []() { server.send_P(200, "text/html; charset=utf-8", UPDATE_FORM); });
  server.on("/update", HTTP_OPTIONS, []() { cors(); server.send(204, "text/plain", ""); });
  server.on("/update", HTTP_POST, []() {
    cors();
    if (ota_auth_fail) { ota_auth_fail = false; return server.requestAuthentication(BASIC_AUTH, "計步器"); }
    bool ok = !Update.hasError() && Update.isFinished();
    server.send(ok ? 200 : 500, "text/plain; charset=utf-8", ok ? "更新完成，重新開機中…" : String("更新失敗：") + Update.errorString());
    if (ok) { app_flush(); hal_log("[OTA] 重開機"); Serial.flush(); delay(300); ESP.restart(); }
  }, []() {
    HTTPUpload& up = server.upload();
    if (up.status == UPLOAD_FILE_START) {
      if (!server.authenticate("admin", app_web_password())) { ota_auth_fail = true; hal_log("[OTA] 密碼錯誤，拒絕"); return; }
      hal_log("[OTA] 開始接收 %s", up.filename.c_str());
      if (!Update.begin(UPDATE_SIZE_UNKNOWN)) hal_log("[OTA] 無法開始：%s", Update.errorString());
    } else if (ota_auth_fail) {
      return;
    } else if (up.status == UPLOAD_FILE_WRITE) {
      if (Update.write(up.buf, up.currentSize) != up.currentSize) hal_log("[OTA] 寫入錯誤：%s", Update.errorString());
    } else if (up.status == UPLOAD_FILE_END) {
      if (Update.end(true)) hal_log("[OTA] 收完 %u bytes，驗證通過", up.totalSize); else hal_log("[OTA] 失敗：%s", Update.errorString());
    } else if (up.status == UPLOAD_FILE_ABORTED) { Update.abort(); hal_log("[OTA] 中止"); }
  });
}
static void on_request() {
  String uri = server.uri(); String host = server.hostHeader();
  // 不是對我們的 IP 發的請求（例如 captive.apple.com），或是探測網址 → 302 轉到儀表板，手機就會跳出登入頁
  if (is_probe(uri) || (host.length() && host != ip_str && host != "192.168.4.1")) {
    server.sendHeader("Location", String("http://") + ip_str + "/", true); server.send(302, "text/plain", ""); return;
  }
  // 只有「會改資料」的 API 要密碼；儀表板、狀態、時間同步讓連上熱點的人直接用（強制門戶視窗不一定能跳 Basic Auth）
  bool sensitive = (server.method() == HTTP_POST && (uri == "/api/settings" || uri == "/api/reset_steps"));
  if (sensitive && !server.authenticate("admin", app_web_password())) return server.requestAuthentication(BASIC_AUTH, "計步器");
  String body = rebuild_body();
  HttpReq req = { server.method() == HTTP_POST ? "POST" : "GET", uri.c_str(), body.c_str(), true };
  HttpResp r = app_http(req);
  server.sendHeader("Cache-Control", "no-store");
  server.send_P(r.code, r.type, r.body, r.len);
}
void hal_net_begin(const char* ap_ssid, const char* ap_pass, const char* sta_ssid, const char* sta_pass) {
  if (net_active) return;

  WiFi.persistent(false);
  WiFi.mode(WIFI_AP_STA);
  WiFi.softAP(ap_ssid, ap_pass, 1, 0, 4);
  if (sta_ssid && *sta_ssid) {
    hal_log("[網路] 連線到 %s …", sta_ssid);
    WiFi.begin(sta_ssid, sta_pass);
    for (int i = 0; i < 40 && WiFi.status() != WL_CONNECTED; i++) delay(200);
    sta_ok = WiFi.status() == WL_CONNECTED;

  }
  ip_str = (sta_ok ? WiFi.localIP() : WiFi.softAPIP()).toString();
  if (sta_ok && MDNS.begin("pedometer")) { MDNS.addService("http", "tcp", 80); hal_log("[網路] 家用網路 IP %s（也可用 http://pedometer.local/）", ip_str.c_str()); }

  ota_routes();

}

// 序列埠指令（115200）：
//   time <epoch秒> [時區分鐘]   用電腦時間同步（和網頁 POST /api/time 同一條路徑）
//   status                      印出 /api/status 的 JSON
// ---------- 效能統計：每秒的迴圈次數、忙碌時間、最長一圈（序列埠 perf）
static uint32_t pf_loops = 0, pf_busy = 0, pf_max = 0, pf_t0 = 0, pf_all = 0, pf_last_all = 0;
static bool pf_ignore = false;   // 這一圈在跑量測／除錯指令（perf、plog、dump…），不算進統計
static uint32_t pf_last_loops = 0, pf_last_busy = 0, pf_last_max = 0;
static uint32_t pf_last_push = 0, pf_last_skip = 0, pf_last_fifo = 0;
static uint32_t pf_sleep_ms = 0, pf_last_sleep = 0, pf_sleep_total = 0, pf_sleep_count = 0;   // light sleep 毫秒數（每秒／累計）
static uint32_t pf_idle_us = 0;   // 這一圈在 hal_idle 休息了多久（從 CPU 活動時間扣掉）
// plog：每分鐘一筆，存 /plog.bin（拔 USB 跑電池也記得到），最多 4000 筆（約 2.7 天）後重頭
struct PlogRec { uint32_t up_s; uint16_t mv; int16_t temp10; uint16_t busy_ms; uint16_t push; uint8_t flags; uint8_t bl; uint16_t loops; uint16_t sleep_ms; uint16_t steps_min; };
static uint32_t plog_next = 60000;
static void plog_tick(uint32_t now) {
  if (now < plog_next) return; plog_next = now + 60000;
  PlogRec r = { now / 1000, (uint16_t)(hal_battery_volts() * 1000), (int16_t)(temperatureRead() * 10), (uint16_t)(psave ? pf_last_all / 1000 : 1000), (uint16_t)pf_last_push,
                (uint8_t)((psave ? 1 : 0) | (net_active ? 2 : 0) | (scr_power ? 4 : 0)), cur_bl, (uint16_t)(pf_last_loops > 65535 ? 65535 : pf_last_loops), (uint16_t)pf_last_sleep, 0 };
  if (!LittleFS.begin(true)) return;
  File f = LittleFS.open("/plog.bin", "a"); if (!f) return;
  if (f.size() % sizeof(PlogRec)) { f.close(); LittleFS.remove("/plog.bin"); f = LittleFS.open("/plog.bin", "a"); }   // 舊格式
  if (f.size() >= sizeof(PlogRec) * 1500) { f.close(); LittleFS.remove("/plog.bin"); f = LittleFS.open("/plog.bin", "a"); }
  f.write((uint8_t*)&r, sizeof r); f.close();
}
void hal_perf_loop(uint32_t us) {
  if (pf_ignore) { pf_ignore = false; }
  else { uint32_t w = us > pf_idle_us ? us - pf_idle_us : 0; pf_loops++; pf_all += w; if (w > 300) pf_busy += w; if (w > pf_max) pf_max = w; }
  pf_idle_us = 0;   // > 300 µs 才算「有在做事」，其餘是空轉
  uint32_t now = millis();
  if (now - pf_t0 >= 1000) { pf_last_sleep = pf_sleep_ms; pf_sleep_ms = 0; pf_last_fifo = pf_fifo_reads; pf_fifo_reads = 0;
                            pf_last_loops = pf_loops; pf_last_busy = pf_busy; pf_last_max = pf_max; pf_last_push = pf_push; pf_last_skip = pf_skip; pf_last_all = pf_all;
                            pf_loops = pf_busy = pf_max = pf_push = pf_skip = pf_all = 0; pf_t0 = now; }
  plog_tick(now);
}

void hal_idle(uint32_t max_ms, bool sleep_ok) {
  if (!psave) return;
  uint32_t t_in = micros();
  if (max_ms < 1) max_ms = 1;
  if (sleep_ok && !tone_end && max_ms >= 10) {
    esp_sleep_enable_timer_wakeup((uint64_t)max_ms * 1000);
    gpio_wakeup_enable((gpio_num_t)PIN_BTN_L, GPIO_INTR_LOW_LEVEL); gpio_wakeup_enable((gpio_num_t)PIN_BTN_R, GPIO_INTR_LOW_LEVEL);
    esp_sleep_enable_gpio_wakeup();
    uint32_t t0 = millis(); Serial.flush(); esp_light_sleep_start(); uint32_t sl = millis() - t0; pf_sleep_ms += sl; pf_sleep_total += sl; pf_sleep_count++;
    gpio_wakeup_disable((gpio_num_t)PIN_BTN_L); gpio_wakeup_disable((gpio_num_t)PIN_BTN_R);
    esp_sleep_disable_wakeup_source(ESP_SLEEP_WAKEUP_GPIO); esp_sleep_disable_wakeup_source(ESP_SLEEP_WAKEUP_TIMER);
    pf_idle_us += micros() - t_in; return;
  }
  delay(max_ms < 20 ? max_ms : 20);                      // 螢幕開著：最多休息 20 ms，按鍵短按不會漏
  pf_idle_us += micros() - t_in;
}

static uint32_t rng_state = 0x2545F491u;
static void rng_mix(uint32_t v) { rng_state ^= v; rng_state ^= rng_state << 13; rng_state ^= rng_state >> 17; rng_state ^= rng_state << 5; }
static uint32_t rng() { rng_mix(0x9E3779B9u); return rng_state; }

// 甩動偵測：|a| 偏離 1 g 超過門檻算「甩動中」，甩完靜止一段時間才出結果（避免連續出好幾次）
class ShakeDetector {
public:
  // 回傳 true 表示「剛剛甩完、可以出結果」
  bool feed(float mag, uint32_t now) {
    rng_mix((uint32_t)(mag * 100000) ^ now);                     // 用真實雜訊攪亂亂數
    bool strong = fabsf(mag - 1.0f) > 0.6f;
    if (strong) { shaking_ = true; last_strong_ = now; peak_ = fmaxf(peak_, fabsf(mag - 1.0f)); }
    if (shaking_ && now - last_strong_ > 350) { shaking_ = false; float p = peak_; peak_ = 0; return p > 0.6f; }
    return false;
  }
  bool shaking() const { return shaking_; }
private:
  bool shaking_ = false; uint32_t last_strong_ = 0; float peak_ = 0;
};
